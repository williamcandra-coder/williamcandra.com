# Goh Dip Tong pipeline dependencies.
#
# Deliberately minimal. The collectors, validators and writers use only the
# standard library plus these two, so the pipeline runs on a bare Python 3.9+
# with no compiled extensions and no wheels to build.
#
# Not used and not wanted for Stage 1:
#   pandas / pyarrow  - the datasets are small and text-first; a dataframe layer
#                       would add a large dependency for no benefit and make
#                       byte-stable output harder, not easier.
#   lxml / beautifulsoup - no HTML scraping is enabled. If a live HTML source is
#                       ever approved, add the parser dependency then.

PyYAML>=6.0,<7          # config/goh-dip-tong/*.yml
jsonschema>=4.18,<5     # schemas/goh-dip-tong/*.schema.json (Draft 2020-12)

# Live HTTP providers only. Every one of them is disabled in Stage 1, so this is
# imported lazily and is not needed to run the fixture pipeline or the tests.
requests>=2.31,<3

# Test-only.
pytest>=7.4
